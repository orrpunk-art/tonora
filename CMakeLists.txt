cmake_minimum_required(VERSION 3.20)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
  if(NOT CMAKE_OSX_SYSROOT)
    execute_process(COMMAND xcrun --sdk macosx --show-sdk-path OUTPUT_VARIABLE KVARTA_SDK OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    set(CMAKE_OSX_SYSROOT "${KVARTA_SDK}" CACHE PATH "macOS SDK")
  endif()
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
endif()
project(Tonora VERSION 1.2.1 LANGUAGES C OBJC)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(kvarta_core src/music.c src/audio.c src/piano.c src/effects.c src/mixer.c src/performance.c src/midi.c src/fingering.c)
target_include_directories(kvarta_core PUBLIC src)
target_compile_options(kvarta_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(kvarta_core PUBLIC "-framework AudioToolbox" "-framework CoreFoundation")
add_executable(kvarta_icon scripts/icon.m src/brand.m)
target_include_directories(kvarta_icon PRIVATE src)
target_compile_options(kvarta_icon PRIVATE -fobjc-arc)
target_link_libraries(kvarta_icon PRIVATE "-framework Cocoa")
set(KVARTA_ICON "${CMAKE_CURRENT_BINARY_DIR}/Tonora.icns")
add_custom_command(OUTPUT "${KVARTA_ICON}" COMMAND kvarta_icon "${KVARTA_ICON}" DEPENDS kvarta_icon)
set_source_files_properties("${KVARTA_ICON}" PROPERTIES GENERATED TRUE MACOSX_PACKAGE_LOCATION Resources)
add_executable(Tonora MACOSX_BUNDLE src/main.m src/brand.m src/controls.m src/strings_en.m src/midiio.m src/pianoroll.m src/mididrag.m "${KVARTA_ICON}")
target_compile_options(Tonora PRIVATE -fobjc-arc -Wall -Wextra -Wno-missing-field-initializers)
target_link_libraries(Tonora PRIVATE kvarta_core "-framework Cocoa" "-framework CoreMIDI" "-framework UniformTypeIdentifiers")
foreach(SAMPLE_SET IN ITEMS piano CP80 Pianet Strings)
  file(GLOB SAMPLE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/${SAMPLE_SET}/*")
  if(SAMPLE_SET STREQUAL "piano")
    set(RESOURCE_DIR "Piano")
  else()
    set(RESOURCE_DIR "${SAMPLE_SET}")
  endif()
  set_source_files_properties(${SAMPLE_FILES} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${RESOURCE_DIR}")
  target_sources(Tonora PRIVATE ${SAMPLE_FILES})
endforeach()
set_target_properties(Tonora PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/Info.plist")
enable_testing()
add_executable(music_tests tests/music_tests.c src/music.c)
target_include_directories(music_tests PRIVATE src)
target_compile_options(music_tests PRIVATE -UNDEBUG)
add_test(NAME music_theory COMMAND music_tests)
add_test(NAME english_strings COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/tests/strings_check.py" "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_test(NAME native_integration COMMAND $<TARGET_FILE:Tonora> --self-test)

add_executable(piano_tests tests/piano_tests.c src/music.c src/piano.c src/effects.c src/mixer.c src/performance.c)
target_include_directories(piano_tests PRIVATE src)
target_compile_options(piano_tests PRIVATE -UNDEBUG -Wall -Wextra -Wno-missing-field-initializers)
target_link_libraries(piano_tests PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
add_test(NAME piano_render COMMAND piano_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/piano" "${CMAKE_CURRENT_BINARY_DIR}/piano-preview.wav")
add_executable(mixer_tests tests/mixer_tests.c src/music.c src/mixer.c src/piano.c src/effects.c src/performance.c)
target_include_directories(mixer_tests PRIVATE src)
target_compile_options(mixer_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_link_libraries(mixer_tests PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
add_test(NAME realtime_mixer COMMAND mixer_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets/piano" "${CMAKE_CURRENT_SOURCE_DIR}/assets/CP80" "${CMAKE_CURRENT_SOURCE_DIR}/assets/Pianet" "${CMAKE_CURRENT_SOURCE_DIR}/assets/Strings")

set(FX_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/verblib/verblib.h" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/verblib/README.txt")
set_source_files_properties(${FX_LICENSE} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/FX-Licenses")
target_sources(Tonora PRIVATE ${FX_LICENSE})

add_executable(performance_tests tests/performance_tests.c src/performance.c src/music.c)
target_include_directories(performance_tests PRIVATE src)
target_compile_options(performance_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_test(NAME two_hand_patterns COMMAND performance_tests)

add_executable(midi_tests tests/midi_tests.c src/midi.c src/performance.c src/music.c)
target_include_directories(midi_tests PRIVATE src)
target_compile_options(midi_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_test(NAME midi_file COMMAND midi_tests)

add_executable(fingering_tests tests/fingering_tests.c src/fingering.c src/performance.c src/music.c)
target_include_directories(fingering_tests PRIVATE src)
target_compile_options(fingering_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_test(NAME fingering COMMAND fingering_tests)
